# 只允许明确已核的Linux SDK和官方绑定，不在configure时下载或修改依赖。
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
    message(FATAL_ERROR "此桥仅验证Linux-x86_64；不得隐式扩展到其他平台")
endif()
set(HANMO_GODOT_CPP_ROOT "" CACHE PATH "已核官方godot-cpp精确commit目录")
set(HANMO_GODOT_API_ROOT "" CACHE PATH "固定编辑器真实导出的三份SDK文件目录")
string(JSON bridge_state GET "${LOCK_TEXT}" godot_bridge acquisition_state)
if(NOT bridge_state STREQUAL "verified")
    message(FATAL_ERROR "Godot桥依赖未取得或未核摘要")
endif()
string(JSON HANMO_GODOT_CPP_COMMIT GET "${LOCK_TEXT}" godot_bridge binding commit)
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${HANMO_GODOT_CPP_ROOT}"
    OUTPUT_VARIABLE binding_commit OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT binding_commit STREQUAL HANMO_GODOT_CPP_COMMIT)
    message(FATAL_ERROR "godot-cpp commit不符")
endif()
execute_process(COMMAND git status --porcelain --untracked-files=all WORKING_DIRECTORY "${HANMO_GODOT_CPP_ROOT}"
    OUTPUT_VARIABLE binding_changes OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT binding_changes STREQUAL "")
    message(FATAL_ERROR "godot-cpp工作副本有额外/修改源码，不能按固定commit构建")
endif()
string(JSON size LENGTH "${LOCK_TEXT}" godot_bridge binding files)
math(EXPR last "${size}-1")
foreach(index RANGE ${last})
    string(JSON path MEMBER "${LOCK_TEXT}" godot_bridge binding files ${index})
    string(JSON expected GET "${LOCK_TEXT}" godot_bridge binding files "${path}")
    file(SHA256 "${HANMO_GODOT_CPP_ROOT}/${path}" actual)
    if(NOT actual STREQUAL expected)
        message(FATAL_ERROR "godot-cpp源或许可字节不符：${path}")
    endif()
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${HANMO_GODOT_CPP_ROOT}/${path}")
endforeach()
foreach(path extension_api.json gdextension_interface.json gdextension_interface.h)
    string(JSON expected GET "${LOCK_TEXT}" godot_bridge sdk files "${path}")
    file(SHA256 "${HANMO_GODOT_API_ROOT}/${path}" actual)
    if(NOT actual STREQUAL expected)
        message(FATAL_ERROR "固定编辑器真实SDK字节不符：${path}")
    endif()
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${HANMO_GODOT_API_ROOT}/${path}")
endforeach()
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/build_profile.json" HANMO_PROFILE_SHA256)
string(JSON expected_profile GET "${LOCK_TEXT}" godot_bridge profile sha256)
if(NOT HANMO_PROFILE_SHA256 STREQUAL expected_profile)
    message(FATAL_ERROR "只读桥API裁剪profile字节不符")
endif()

# 使用官方CMake完成生成；异常保持启用，ABI异常被自有桥边界接住。
set(GODOTCPP_CUSTOM_API_FILE "${HANMO_GODOT_API_ROOT}/extension_api.json" CACHE FILEPATH "" FORCE)
set(GODOTCPP_GDEXTENSION_DIR "${HANMO_GODOT_API_ROOT}" CACHE PATH "" FORCE)
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/build_profile.json" CACHE PATH "" FORCE)
# 固定编辑器需要debug特性ABI；CMake Release仅表示编译优化，不能替代Godot target。
set(GODOTCPP_TARGET "template_debug" CACHE STRING "" FORCE)
set(GODOTCPP_PRECISION "single" CACHE STRING "" FORCE)
set(GODOTCPP_DISABLE_EXCEPTIONS OFF CACHE BOOL "" FORCE)
set(GODOTCPP_USE_HOT_RELOAD OFF CACHE BOOL "" FORCE)
set(GODOTCPP_ENABLE_TESTING OFF CACHE BOOL "" FORCE)
add_subdirectory("${HANMO_GODOT_CPP_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/binding")
file(SHA256 "${HANMO_GODOT_API_ROOT}/extension_api.json" HANMO_API_SHA256)
file(SHA256 "${HANMO_GODOT_API_ROOT}/gdextension_interface.json" HANMO_INTERFACE_SHA256)
file(SHA256 "${CMAKE_CURRENT_BINARY_DIR}/binding/extension_api.json" HANMO_TRIMMED_API_SHA256)

set(identity "")
foreach(path CMakeLists.txt exports.map build_profile.json hanmo_world_bridge.hpp hanmo_world_bridge.cpp register_types.cpp bridge_identity.hpp.in hanmo_world.gdextension.in)
    file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${path}" sha)
    string(APPEND identity "${path}:${sha}\n")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${path}")
endforeach()
string(SHA256 HANMO_BRIDGE_SOURCE_SHA256 "${identity}")
configure_file(bridge_identity.hpp.in generated/bridge_identity.hpp @ONLY)
add_library(hanmo_world_bridge SHARED hanmo_world_bridge.cpp register_types.cpp)
target_include_directories(hanmo_world_bridge PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated" "${CMAKE_CURRENT_SOURCE_DIR}/../world/src")
target_link_libraries(hanmo_world_bridge PRIVATE hanmo_world godot::cpp)
set_target_properties(hanmo_world_bridge PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(hanmo_world_bridge PRIVATE -Wall -Wextra)
    # Linux/ELF静态依赖只留在本桥；禁止C++ locale/唯一符号与Mesa等宿主依赖混用。
    target_link_options(hanmo_world_bridge PRIVATE "LINKER:--exclude-libs,ALL" "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
    # 同时约束直接对象的标准库RTTI导出；改导出表也必须重链并刷新源码身份。
    set_property(TARGET hanmo_world_bridge APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
endif()
